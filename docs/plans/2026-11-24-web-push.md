# Web Push Implementation Plan

> **For Claude:** Use `${SUPERPOWERS_SKILLS_ROOT}/skills/collaboration/executing-plans/SKILL.md` to implement this plan task-by-task.

**Goal:** Add standards-based browser Web Push to Planit so dossier submission and correction-request notifications reach users when the tab is closed, without replacing the existing SSE/Supabase Realtime in-app transport.

**Architecture:** Keep `notifications` as the source of truth for in-app notifications. Add VAPID-authenticated subscriptions and a durable delivery table, then hook the existing dossier workflow notification insert to enqueue Web Push deliveries and drain them asynchronously. The service worker renders and routes `push` events; native APNs/FCM, OneSignal, silent pushes and provider SDKs are out of scope.

**Tech Stack:** Next.js 16.4.6 App Router, TypeScript 5.9.3, Drizzle ORM + PostgreSQL, Vitest 5.1.1, `web-push` 3.x, VAPID P-256 keys, Orca CLI for browser validation.

## Goal

Un utilisateur authentifié qui a activé les notifications navigateur reçoit une notification système pour une soumission de dossier ou une demande de corrections, même si l'onglet Planit est fermé, sans duplicate et sans casser le flux SSE existant.

## Scope

**In scope**
- Génération et configuration des clés VAPID via variables d'environnement.
- Table `push_subscriptions` et table `notification_deliveries`.
- Routes authentifiées d'abonnement/désabonnement et route publique de clé publique.
- Helper client `PushManager.subscribe` et bouton d'activation dans l'UI.
- Handlers `push` et `notificationclick` dans `public/sw.js`.

**Out of scope**

// lib/web-push/config.ts
export interface WebPushConfig {
  publicKey: string;
  privateKey: string;
  subject: string;
}

export function getWebPushConfig(): WebPushConfig | null {
  const publicKey = process.env.NEXT_PUBLIC_VAPID_PUBLIC_KEY?.trim();
  const privateKey = process.env.WEB_PUSH_VAPID_PRIVATE_KEY?.trim();
  const subject = process.env.WEB_PUSH_VAPID_SUBJECT?.trim();

  if (!publicKey || !privateKey || !subject) return null;
  return { publicKey, privateKey, subject };
}

export function isWebPushConfigured(): boolean {
  return getWebPushConfig() !== null;
}

export function requireWebPushConfig(): WebPushConfig {
  const config = getWebPushConfig();
  if (!config) throw new Error('WEB_PUSH_NOT_CONFIGURED');
  return config;
}
```

// lib/web-push/validation.ts
export interface PushSubscriptionInput {
  endpoint: string;
  p256dh: string;
  auth: string;
}

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null;
}

export function parsePushSubscriptionInput(body: unknown): PushSubscriptionInput | null {
  if (!isRecord(body)) return null;

  const endpoint = body.endpoint;
  const keys = body.keys;
  if (typeof endpoint !== 'string' || !isRecord(keys)) return null;

  const p256dh = keys.p256dh;
  const auth = keys.auth;
  if (typeof p256dh !== 'string' || typeof auth !== 'string') return null;
  if (!endpoint || !p256dh || !auth) return null;
  if (endpoint.length > 2048 || p256dh.length > 512 || auth.length > 512) return null;

  let parsed: URL;
  try {
    parsed = new URL(endpoint);
  } catch {
    return null;
  }
  if (parsed.protocol !== 'https:') return null;

  return { endpoint, p256dh, auth };
}
```

import { and, eq } from 'drizzle-orm';

import { getAuthenticatedUser } from '@/lib/auth-server';

export async function POST(request: NextRequest) {
  const user = await getAuthenticatedUser();
  if (!user) return NextResponse.json({ error: 'Non autorisé' }, { status: 401 });

  const body: unknown = await request.json();
  const parsed = parsePushSubscriptionInput(body);
  if (!parsed) return NextResponse.json({ error: 'Abonnement invalide' }, { status: 400 });

  const db = getDrizzleDb();
  if (!db || !isDatabaseConfigured()) {
    return NextResponse.json({ error: 'Base non configurée' }, { status: 503 });
  }
  const setup = await ensureDatabaseTables();
  if (!setup.success) return NextResponse.json({ error: 'Base indisponible' }, { status: 503 });

  const existing = await db
    .select()
    .from(pushSubscriptions)
    .where(eq(pushSubscriptions.endpoint, parsed.endpoint))
    .limit(1);

  const current = existing[0];
  if (current && current.userId !== user.id) {
    return NextResponse.json({ error: 'Endpoint déjà associé' }, { status: 409 });
  }

  const userAgent = request.headers.get('user-agent');

  await db
    .insert(pushSubscriptions)
    .values({
      id: `push-sub-${randomUUID()}`,
      userId: user.id,
      endpoint: parsed.endpoint,
      p256dh: parsed.p256dh,
      auth: parsed.auth,
      userAgent,
      failureCount: 0,
      lastStatusCode: null,
      lastError: null,
      updatedAt: new Date(),
    })
    .onConflictDoUpdate({
      target: pushSubscriptions.endpoint,
      set: {
        userId: user.id,
        p256dh: parsed.p256dh,
        auth: parsed.auth,
        userAgent,
        failureCount: 0,
        lastStatusCode: null,
        lastError: null,

// lib/web-push/client.ts
export type SubscribeResult =
  | 'subscribed'
  | 'already-subscribed'
  | 'denied'
  | 'unsupported'
  | 'failed';

export function urlBase64ToUint8Array(value: string): Uint8Array {
  const padding = '='.repeat((4 - (value.length % 4)) % 4);
  const base64 = (value + padding).replace(/-/g, '+').replace(/_/g, '/');
  const raw = atob(base64);
  const output = new Uint8Array(raw.length);
  for (let index = 0; index < raw.length; index += 1) {
    output[index] = raw.charCodeAt(index);
  }
  return output;
}

export async function subscribeCurrentBrowser(): Promise<SubscribeResult> {
  if (
    typeof window === 'undefined' ||
    !('serviceWorker' in navigator) ||
    !('PushManager' in window) ||
    typeof Notification === 'undefined'
  ) {
    return 'unsupported';
  }

  const publicKeyResponse = await fetch('/api/push/public-key');
  const publicKeyPayload = (await publicKeyResponse.json()) as {
    enabled?: boolean;
    publicKey?: string;
  };
  if (!publicKeyPayload.enabled || !publicKeyPayload.publicKey) return 'unsupported';

  if (Notification.permission === 'denied') return 'denied';
  if (Notification.permission === 'default') {
    const permission = await Notification.requestPermission();
    if (permission !== 'granted') return 'denied';
  }

  const registration = await navigator.serviceWorker.ready;
  const existing = await registration.pushManager.getSubscription();
  const subscription =
    existing ||
    (await registration.pushManager.subscribe({
      userVisibleOnly: true,
      applicationServerKey: urlBase64ToUint8Array(publicKeyPayload.publicKey),
    }));

  if (existing) return 'already-subscribed';

  const response = await fetch('/api/push/subscribe', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(subscription.toJSON()),
  });

  return response.ok ? 'subscribed' : 'failed';
}

// lib/web-push/sender.ts
import webPush from 'web-push';
import { requireWebPushConfig } from '@/lib/web-push/config';
import type { WebPushPayload } from '@/lib/web-push/payload';

export interface PushTarget {
  endpoint: string;
  p256dh: string;
  auth: string;
}

export interface WebPushSendResult {
  statusCode: number | null;
  expired: boolean;
  retryAt: number | null;
  error: string | null;
}

interface WebPushErrorShape {
  statusCode?: number;
  headers?: Record<string, string | undefined>;
  message?: string;
}

let configured = false;

function configure(): void {
  if (configured) return;
  const config = requireWebPushConfig();
  webPush.setVapidDetails(config.subject, config.publicKey, config.privateKey);
  configured = true;
}

function readRetryAfter(headers: Record<string, string | undefined> | undefined): number | null {
  const value = headers?.['retry-after'];
  if (!value) return null;
  const seconds = Number(value);
  if (Number.isFinite(seconds)) return Date.now() + seconds * 1000;
  const date = Date.parse(value);
  return Number.isFinite(date) ? date : null;
}

export async function sendWebPush(
  target: PushTarget,
  payload: WebPushPayload
): Promise<WebPushSendResult> {
  configure();

  try {
    const response = await webPush.sendNotification(target, JSON.stringify(payload), {
      TTL: 86400,
      urgency: 'high',
      topic: payload.tag,
    });
    return { statusCode: response.statusCode ?? null, expired: false, retryAt: null, error: null };
  } catch (caught: unknown) {
    const error = caught as WebPushErrorShape;
    const statusCode = typeof error.statusCode === 'number' ? error.statusCode : null;
    const expired = statusCode === 404 || statusCode === 410;
    const retryAt = statusCode === 429 ? readRetryAfter(error.headers) : null;
    return {
      statusCode,
      expired,
      retryAt,
      error: error.message || 'WEB_PUSH_SEND_FAILED',
    };
  }
}

export interface DeliverySummary {
  sent: number;
  failed: number;
  expired: number;
  retried: number;
}

interface DeliveryOptions {
  recipientId?: string;
  limit?: number;
}

function retryDelay(attempts: number, override: number | null): number {
  if (override !== null) return override;
  return Math.min(2 ** Math.max(0, attempts) * 60_000, 24 * 60 * 60 * 1000);
}

export async function enqueueWebPushDeliveries(
  notification: TaskNotification

  if (!isWebPushConfigured()) return [];
  const db = getDrizzleDb();
  if (!db || !isDatabaseConfigured()) return [];
  const setup = await ensureDatabaseTables();
  if (!setup.success) return [];

  const subscriptions = await db
    .select()
    .from(pushSubscriptions)
    .where(eq(pushSubscriptions.userId, notification.targetUserId || ''))
    .limit(50);

  const payload: WebPushPayload = buildWebPushPayload(notification);
  const ids: string[] = [];

  for (const subscription of subscriptions) {
    const id = `push-del-${randomUUID()}`;
    await db
      .insert(notificationDeliveries)
      .values({
        id,
        notificationId: notification.id,
        recipientId: subscription.userId,
        channel: 'web_push',
        dedupeKey: `${notification.id}:${subscription.id}`.slice(0, 160),
        status: 'pending',
        attempts: 0,
        maxAttempts: 5,
        nextAttemptAt: new Date(),
        sentAt: null,
        lastStatusCode: null,
        lastError: null,
        payload,
        updatedAt: new Date(),
      })
      .onConflictDoNothing();
    ids.push(id);
  }

  return ids;
}

export async function processPendingWebPushDeliveries(
  options: DeliveryOptions = {}
): Promise<DeliverySummary> {
  const summary: DeliverySummary = { sent: 0, failed: 0, expired: 0, retried: 0 };
  if (!isWebPushConfigured()) return summary;

  const db = getDrizzleDb();
  if (!db || !isDatabaseConfigured()) return summary;
  const setup = await ensureDatabaseTables();
  if (!setup.success) return summary;

  const limit = Math.min(Math.max(options.limit || 20, 1), 50);
  const pending = await db
    .select({ delivery: notificationDeliveries, subscription: pushSubscriptions })
    .from(notificationDeliveries)
    .innerJoin(pushSubscriptions, eq(notificationDeliveries.recipientId, pushSubscriptions.userId))
    .where(
      and(
        eq(notificationDeliveries.status, 'pending'),
        lte(notificationDeliveries.nextAttemptAt, new Date())
      )
    )
    .orderBy(asc(notificationDeliveries.createdAt))
    .limit(limit);

  for (const row of pending) {
    const attempts = row.delivery.attempts + 1;
    const result = await sendWebPush(row.subscription, row.delivery.payload);

    if (result.expired) {
      await db
        .delete(pushSubscriptions)
        .where(eq(pushSubscriptions.id, row.subscription.id));
      await db
        .update(notificationDeliveries)
        .set({
          status: 'expired',
          attempts,
          lastStatusCode: result.statusCode,
          lastError: result.error,
          updatedAt: new Date(),
        })
        .where(eq(notificationDeliveries.id, row.delivery.id));
      summary.expired += 1;
      continue;
    }

    const shouldRetry =
      result.retryAt !== null || (result.statusCode !== null && result.statusCode >= 500);
    if (shouldRetry && attempts < row.delivery.maxAttempts) {
      await db
        .update(notificationDeliveries)
        .set({
          status: 'pending',
          attempts,
          nextAttemptAt: new Date(retryDelay(attempts, result.retryAt)),
          lastStatusCode: result.statusCode,
          lastError: result.error,
          updatedAt: new Date(),
        })
        .where(eq(notificationDeliveries.id, row.delivery.id));
      summary.retried += 1;
      continue;
    }

    const succeeded = result.statusCode !== null && result.statusCode < 300;
    await db
      .update(notificationDeliveries)
      .set({
        status: succeeded ? 'sent' : 'failed',
        attempts,
        nextAttemptAt: new Date(),
        sentAt: succeeded ? new Date() : null,
        lastStatusCode: result.statusCode,
        lastError: succeeded ? null : result.error,
        updatedAt: new Date(),
      })
      .where(eq(notificationDeliveries.id, row.delivery.id));

    if (succeeded) summary.sent += 1;
    else summary.failed += 1;
  }

  return summary;
}
